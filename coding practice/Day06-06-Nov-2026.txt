--> K8s - POD with volume mount

kind: pod
metadata:
	name: day6-pod
spec:
  containers:
	- name: web
	  image: nginx
	  volumeMounts:
		- mountPath: /usr/share/nginx/pod
		  name: pod-volume
  volumes:
	- name: html-volume
	  hostPath:
		path: /data
		type: directory

--> Shell script to check running services

#!/bin/bash
services=("nginx" "docker" "httpd")
for services in "${services[@]}"
do
  if systemctl is-active --quiet $service; then
	echo "$service is running"
  else
	echo "$service is not running"
  fi
done
