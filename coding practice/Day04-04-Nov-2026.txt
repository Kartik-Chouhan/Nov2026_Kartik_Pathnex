Day 04 - basic CI/CD concepts with Jenkins and gitlab

--> Jenkins - basic CI pipeline

pipeline{
	agent any
	stages{
		stage('Build'){
			steps{
				echo "Project is getting build.."
			}
		}
		stage('Test'){
			steps{
				echo "Project is in testing phase... Test cases are running.."
			}
		}
		stage('Deploy'){
			steps{
				echo "Deploying the application..."
			}
		}
	}
}

--> Gitlab CI/CD -- basic pipeline

stages:
	- build
	- test
	- deploy
	
build:
	stage: build
	script:
	  - echo "Building project.."
	  
test:
	stage: test
	script:
	  - echo "Testing the cases"
	  
deploy:
	stage: deploy
	script:
	  - echo "Deploying the application"
	  
--> Ansible -- install and configure the nginx

- name: install and configure Nginx
  hosts: all
  become: yes
  tasks:
	- name: Install nginx
	  yum: 
	    name: Nginx
	    state: present
	
	- name: Configure nginx
	  template:
		src: nginx.conf.j3
		dest: /etc/nginx/nginx.conf
	
	- name: Start nginx service
	  service:
		name: nginx
		state: present
		enabled: yes
