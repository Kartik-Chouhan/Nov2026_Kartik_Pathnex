Day 07 - Automating cloud infra with terraform & ansible

--> Ansible - setuop Nginx with custom config

- name: setup nginx with custom config
  hosts: all
  become: yes
  tasks:
	- name: install nginx
	  yum:
	  name: nginx
	  state: present
	- name: copy custom nginx.conf
	  copy:
		src: nginx.conf
		dest: /etc/nginx/nginx.conf
	- name: start nginx service
	  service:
		name: nginx
		state: started
		enabled: yes	
		
--> Terraform - provision EC2 with IAM role

resource "aws_iam_role" "ec2_role" {
	name = "pathnex_ec2_role"
	assume_role_policy = jsonencode({
	  version = "2020-10-20"
	  statement = [
		{
			Action = "sts.AssumeRole"
			Effect = "Allow"
			Principal = {
				service = "ec2.amazonaws.com"
			}
		},
	  ]
	})
}

resource "aws_instance" "kartik_ec2"{
	ami = "ami_0abcd12134ABCD1234"
	instance_type = "t2.medium"
	iam_instance_profile = aws_iam_role.ec2_role.name
	tags = {
	   Name = "Kartik-EC2"
	}
}
