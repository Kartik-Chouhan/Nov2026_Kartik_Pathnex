Day03 - Ansible, Terraform & K8s basics

--> Ansible - Install Apache and start service

- name: install and start apache service
  hosts: all
  become: yes
  
  tasks:
	- name: install apache
	  yum: 
		name: http
		state: present
		
	- name: start apache
	  service:
		name: http
		state: started
		enabled: yes
		
--> Terraform - EC2 Instance(t3.medium)

provider "aws"{
	region = "us-east-1"
}

resource "aws-instance" "Pathnex-EC2"{
	ami = "ami-0abcd1234abcd1234"
	intance_type = "t3.medium"
	
	tags={
		name = "Pathnex-T3"
	}
}

--> K8s - create a ReplicaSet with 3 replicas

apiVersion: apps/v2
kind: ReplicaSet
metadata:
	name: pathnex-replicaset
spec:
	replicas: 3
	selector:
		matchLabels:
			app: pathnex-app
	template:
		metadata:
			labels:
				app: pathnex-app
		spec:
		  containers:
			- name: app
			  image: nginx
			  ports:
				- containerPort: 80
